# P9754 CSP-S 2023 T3 结构体

## 题意

模拟 C 风格结构体的内存对齐。四种操作：定义结构体类型（输出大小和对齐）、定义元素（输出起始地址）、访问 `a.b.c`（输出地址）、访问内存地址（输出占用它的基本类型元素路径，否则 `ERR`）。$n \le 100$，地址到 $10^{18}$。

**标签**：大模拟、字符串　**难度**：5/10

## 解题思路

### 1. 底层逻辑

所有操作都基于两个信息：
1. 每个类型的**大小、对齐要求、成员列表**（成员类型、名称、偏移）；
2. 全局元素列表（类型、名称、起始地址）。

### 2. 隐藏条件

- 对齐公式统一：把 $x$ 向上取整到 $a$ 的倍数，$\lceil x/a \rceil \cdot a$。
  - 成员偏移：上一个成员结束位置向上对齐到本成员的对齐要求；
  - 结构体对齐 = 成员对齐的最大值；大小 = 最后一个成员结束位置向上对齐到结构体对齐；
  - 元素地址：上一个元素结束位置向上对齐。
- 结构体只能引用先前定义的类型，所以定义时成员的大小都已知，一次计算即可。
- 操作 4 从元素逐层往下找包含该地址的成员；若落在**填充（padding）**里就是 `ERR`。

### 3. 注意事项

- 地址、大小可达 $10^{18}$，用 64 位。
- 不同结构体可以有同名成员，成员名只在所属结构体内查找。
- 4 种基本类型可以统一当作"没有成员的类型"，编号 1~4，大小与对齐分别为 1、2、4、8。
- 操作 3 的路径用 `.` 切分，`stringstream` + `getline(ss, part, '.')` 很方便。

### 4. 解决方案

- 操作 1：依次读成员，`off = up(off, al[v])` 得到偏移并记录，`off += sz[v]`，更新对齐；最后 `sz = up(off, al)`。
- 操作 2：`addr = up(top, al[v])`，`top = addr + sz[v]`。
- 操作 3：先找到元素得到类型和地址，再逐级在成员中找名字，地址累加偏移。
- 操作 4：找到包含 `addr` 的元素（找不到为 `ERR`），然后当类型不是基本类型时，在成员中找满足 `偏移 <= addr < 偏移 + 大小` 的成员，找不到即 `ERR`；找到则路径追加成员名，`addr` 减去偏移，进入该成员类型。

数据量很小，暴力查找即可。

## 部分分策略 / 错误思路

- 性质 A（没有结构体）：只需处理基本类型元素。
- 性质 C（结构体成员都是基本类型）：只有一层嵌套。
- 常见错误：结构体大小忘了末尾对齐；操作 4 落在填充区时输出了某个成员。

## AC 代码

```cpp
#include <bits/stdc++.h>
using namespace std;

#define endl '\n'
#define pb push_back
#define F first
#define S second
#define For(i, a, b) for (int i = (a); i <= (b); i++)
#define Rep(i, a, b) for (int i = (a); i < (b); i++)
#define mset(arr, x) memset(arr, x, sizeof(arr))
#define dbg(x) cout << #x << " = " << x << endl

#define int long long
typedef long long ll;
typedef unsigned long long ull;
typedef pair<int, int> pii;
typedef pair<ll, ll> pll;

const int INF = 0x3f3f3f3f;
const ll LINF = 0x3f3f3f3f3f3f3f3fLL;
const double EPS = 1e-9;
const int MOD = 1e9 + 7;

const int N = 205;
int n, tot, cnt, top;
int sz[N], al[N];
vector<int> mt[N], mo[N];
vector<string> mn[N];
map<string, int> tid;
int et[N], ea[N];
string en[N];

inline int up(int x, int a) { return (x + a - 1) / a * a; }

void solve() {
    string base[4] = {"byte", "short", "int", "long"};
    Rep(i, 0, 4) tot++, tid[base[i]] = tot, sz[tot] = al[tot] = 1 << i;
    cin >> n;
    while (n--) {
        int op;
        cin >> op;
        if (op == 1) {
            string s;
            int k, off = 0;
            cin >> s >> k;
            int u = ++tot;
            tid[s] = u;
            For(i, 1, k) {
                string t, nm;
                cin >> t >> nm;
                int v = tid[t];
                off = up(off, al[v]);
                mt[u].pb(v), mn[u].pb(nm), mo[u].pb(off);
                off += sz[v], al[u] = max(al[u], al[v]);
            }
            sz[u] = up(off, al[u]);
            cout << sz[u] << " " << al[u] << endl;
        } else if (op == 2) {
            string t, nm;
            cin >> t >> nm;
            int v = tid[t];
            cnt++, et[cnt] = v, en[cnt] = nm, ea[cnt] = up(top, al[v]);
            top = ea[cnt] + sz[v];
            cout << ea[cnt] << endl;
        } else if (op == 3) {
            string s, part;
            cin >> s;
            stringstream ss(s);
            getline(ss, part, '.');
            int u = 0, addr = 0;
            For(i, 1, cnt) if (en[i] == part) u = et[i], addr = ea[i];
            while (getline(ss, part, '.'))
                Rep(i, 0, mt[u].size()) if (mn[u][i] == part) {
                    addr += mo[u][i], u = mt[u][i];
                    break;
                }
            cout << addr << endl;
        } else {
            int addr, id = 0;
            cin >> addr;
            For(i, 1, cnt) if (ea[i] <= addr && addr < ea[i] + sz[et[i]]) id = i;
            if (!id) {
                cout << "ERR" << endl;
                continue;
            }
            string res = en[id];
            int u = et[id];
            addr -= ea[id];
            bool ok = true;
            while (u > 4) {
                int nxt = -1;
                Rep(i, 0, mt[u].size()) if (mo[u][i] <= addr && addr < mo[u][i] + sz[mt[u][i]]) nxt = i;
                if (nxt == -1) {
                    ok = false;
                    break;
                }
                res += "." + mn[u][nxt], addr -= mo[u][nxt], u = mt[u][nxt];
            }
            cout << (ok ? res : "ERR") << endl;
        }
    }
}

signed main() {
    ios::sync_with_stdio(false), cin.tie(nullptr);
    // freopen("xxx.in", "r", stdin);
    // freopen("xxx.out", "w", stdout);
    int tests = 1;
    // cin >> tests;
    while (tests--) solve();
    return 0;
}
```
